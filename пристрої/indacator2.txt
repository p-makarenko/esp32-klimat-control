// --- ЛОГІКА ESP32 ---

void appendHistory(float energy) {
    if (isnan(energy) || energy <= 0) return;
    File f = LittleFS.open("/history.csv", "a");
    if (f) {
        time_t now; time(&now);
        f.printf("%lu,%.3f\n", now, energy);
        f.close();
    }
}

void updateSensors() {
    float v = pzem.voltage();
    if (!isnan(v)) {
        data.v = v;
        data.c = pzem.current();
        data.p = pzem.power();
        data.e = pzem.energy();
        data.err = false;

        struct tm ti;
        if (getLocalTime(&ti)) {
            if (currentHour != ti.tm_hour) {
                if (currentHour != -1) appendHistory(data.e);
                currentHour = ti.tm_hour;
            }
        }
    } else { data.err = true; }
}

void setup() {
    Serial.begin(115200);
    LittleFS.begin(true);

    WiFi.begin(ssid, password);
    while (WiFi.status() != WL_CONNECTED) delay(500);

    configTime(7200, 3600, ntpServer); // GMT+2 Україна

    server.on("/", HTTP_GET, [](AsyncWebServerRequest *r){
        r->send_P(200, "text/html", index_html);
    });

    server.on("/data", HTTP_GET, [](AsyncWebServerRequest *r){
        String j;
        StaticJsonDocument<128> doc;
        doc["v"] = data.v; doc["c"] = data.c; doc["p"] = data.p; doc["e"] = data.e; doc["err"] = data.err;
        serializeJson(doc, j);
        r->send(200, "application/json", j);
    });

    server.on("/history", HTTP_GET, [](AsyncWebServerRequest *r){
        r->send(LittleFS, "/history.csv", "text/plain");
    });

    server.begin();
}

void loop() {
    if (millis() - lastUpdate > 2000) {
        updateSensors();
        lastUpdate = millis();
        // Реконект WiFi
        if (WiFi.status() != WL_CONNECTED) WiFi.reconnect();
    }
}
